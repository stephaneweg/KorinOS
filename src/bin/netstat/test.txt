# le pid des programmes (colonne Programme) dépend du démarrage : remplacé par « PID » (fonction p, en bash) ;
# l'ordre des sockets aussi (services lancés en parallèle) : lignes triées
p() { while IFS= read -r l; do case "$l" in tcp*|udp*) case "$l" in *"(netd)") echo "$l";; *) r="${l% *}"; echo "${r% [0-9]*} PID ${l##* }";; esac;; *) echo "$l";; esac; done; }
