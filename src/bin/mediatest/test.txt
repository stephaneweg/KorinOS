# mediatest : la ligne d'arguments brute. Les durées (« N images en T ms : F images/s »,
# « décodage seul … us », « … kHz en T ms ») dépendent du temps : chiffres retirés (fonction f).
f() { while IFS= read -r l; do case "$l" in *" ms : "*|*seul*|*"kHz en"*) echo "$l" | tr -d 0-9;; *) echo "$l";; esac; done; }
