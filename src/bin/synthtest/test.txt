# synthtest : les essais en mémoire (FM, banque FM, SoundFont, .mid, .fms) ; -p joue un fichier
# en temps réel : les positions (ms) dépendent du temps, chiffres retirés (tr -d). Les poignées
# (« SoundFont lue », « ouverture ») sont des adresses du tas : chiffres retirés (fonction f).
f() { while IFS= read -r l; do case "$l" in *lue\ =*|*ouverture\ =*) echo "$l" | tr -d 0-9;; *) echo "$l";; esac; done; }
synthtest | f
