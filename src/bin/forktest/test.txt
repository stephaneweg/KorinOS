# forktest : les essais (fork brut, fichier partagé sur /tmp et /forktest.tmp, effacé ensuite),
# 1000 puis 20 fork + exit dans la boucle de fuites. Durée et pages perdues masquées (mesures).
forktest > /tmp/w/o.txt; c=$?; sed 's/ en [0-9]* ms ; pages libres perdues : .*/ en T ms ; pages libres perdues : X/' /tmp/w/o.txt; /bin/echo "code $c"
forktest 20 > /tmp/w/o.txt; c=$?; sed 's/ en [0-9]* ms ; pages libres perdues : .*/ en T ms ; pages libres perdues : X/' /tmp/w/o.txt; /bin/echo "code $c"
ls /forktest.tmp /tmp/forktest.tmp
