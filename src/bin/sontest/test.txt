# sontest : la ligne d'arguments brute ; QEMU joue sur la sortie virtuelle, en temps réel. Les
# lignes de comptes (trames, durées, cadence) dépendent du temps : chiffres retirés (fonction f).
# Le faux pilote et le faux micro s'inscrivent auprès de sound (essais à garder en dernier).
f() { while IFS= read -r l; do case "$l" in *trames*) echo "$l" | tr -d 0-9;; *) echo "$l";; esac; done; }
sontest | f
sontest > RAM0:/tmp/w/s.txt; /bin/echo "code $?"; f < RAM0:/tmp/w/s.txt
sontest 22050 | f
sontest 3000 | f
sontest x | f
sh -c 'sontest 8000' | f
sontest mic 16000 1 | f
sontest driver 48000 2 | f
