# Services lances par init au demarrage du PC (module init.txt de l'initrd, mis
# par tools/pc/build.sh), avant /, qui est sur le disque NVMe lui-meme.
# Ligne : nom d'un module de l'initrd, puis ses arguments.
#
# le bus PCIe, sans annonce (hwmanager n'est pas encore lance ; init2 demande a
# pcid de tout annoncer de nouveau des que hwmanager tourne)
pcid -n
# le disque NVMe : ecriture permise sur la partition KORIN OS seulement (regles
# de nvme.txt dans l'initrd), servie aussi en blk.korin
nvme
# attendre que le pilote publie blk.korin (sur le vrai disque, la remise a zero
# du controleur prend plusieurs secondes ; fatfs n'attend son peripherique que 3 s)
wait blk.korin 30000
# / = la partition KORIN OS : la racine du systeme
vfs / blk.korin fatfs
