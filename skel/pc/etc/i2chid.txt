# Reglages du touchpad I2C (i2chid, lance a la main : i2chid probe | scan | info | run | stop)
# docs/etude/TOUCHPAD-PC.md, « Ce que disent les tables du ZBook ». Une « cle=valeur » sur la ligne de
# commande remplace celle d'ici pour une fois.
#
# --- HP ZBook Fury G8 (DSDT lu le 13/11) : touchpad TPD0 (ELAN0750, _CID PNP0C50) sous I2C0
# controleur I2C0 : _ADR 0x00150000 (sur)
pci = 0000:00:15.0
# adresse I2C : _INI de TPD0, BADR = 0x2C (sur : ecrite en dur, la meme pour tous les modeles)
addr = 0x2c
# registre du descripteur HID : _DSM HID over I2C, fonction 1 -> HID2 = 0x20 (_INI) (sur)
hid_desc = 0x20
# vitesse : I2cSerialBusV2 ... 0x00061A80 = 400000 Hz (sur)
speed = 400
# BAR0 : le _PS3 du controleur (SOD3) la met a 0 avec le passage en D3 ; aucune adresse prevue dans
# l'ACPI en mode PCI. auto : 0x6053ba6000 d'apres le journal pcid ; celle de Windows (Gestionnaire de
# peripheriques, ressources du 43E8) est plus sure
bar = auto
# interruption : GpioInt (Level, ActiveLow) sur \_SB.GPI0 quand TPDM = 0, ce que confirme Windows
# (IRQ 1024 : interruption de GPIO) ; le plot vient de GPDI. Le plot est lu toutes les « poll » ms (bit
# RX de son DW0), le touchpad seulement s'il demande ; gpio_low = -1 : polarite mesuree au repos
poll = 8
gpio_low = -1
# valeurs lues a l'execution dans la NVS du micrologiciel (decalages des champs de ce DSDT : surs pour
# ce BIOS ; la region est retrouvee dans le DSDT vivant). Elles donnent le plot (gpio), et HCNT / LCNT /
# maintien de SDA du FMCN de I2C0 (si USTP) ; un gpio=, irq= ou hcnt= ecrit ici passe avant
nvs_gpdi = GNVS+0x479
nvs_tpdm = GNVS+0x40D
nvs_sbrg = PNVA+0x9C
nvs_pchs = PNVA+0x0
nvs_fmcn = GNVS+0x226
nvs_ustp = GNVS+0x43B
# horloge d'entree du controleur (MHz) : pour le calcul sans FMCN
#clock_mhz = 100
# auto, mouse ou ptp ; vitesse du pointeur (%), toucher pour cliquer, defilement naturel,
# course de deux doigts pour un cran de molette (centiemes de mm)
#mode = auto
#pointer = 100
#tap = 1
#natural = 1
#scroll = 300
